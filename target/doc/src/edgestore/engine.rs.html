18/// Compute exclusive upper bound for a prefix scan: increment last non-0xFF byte.
19fn prefix_upper_bound(prefix: &[u8]) -> Option<Vec<u8>> {
20    let mut end = prefix.to_vec();
21    for i in (0..end.len()).rev() {
22        if end[i] < 0xFF {
23            end[i] += 1;
24            end.truncate(i + 1);
25            return Some(end);
26        }
27        end[i] = 0;
28    }
29    None
30}

48    pub fn open(config: EdgestoreConfig) -> Result<Engine, EdgestoreError> {
49        std::fs::create_dir_all(&config.path)?;
50
51        let lock_path = config.path.join("LOCK");
52        let lockfile = std::fs::OpenOptions::new()
53            .create(true)
54            .write(true)
55            .truncate(false)
56            .open(&lock_path)?;
57
58        lockfile.try_lock_exclusive().map_err(|e| {
59            if e.kind() == std::io::ErrorKind::WouldBlock {
60                EdgestoreError::WriterBusy
61            } else {
62                EdgestoreError::Io(e)
63            }
64        })?;
65
66        let mut memtable = (config.memtable_factory)();
67
68        // Run recovery — replay all WAL files into the memtable.
69        let result = crate::recovery::recover_from_wal(&config.path, &mut memtable)?;
70        let lsn_counter = result.max_lsn;
71        let txid_counter = result.max_txid;
72
73        let wal_files = crate::recovery::list_wal_files(&config.path)?;
74
75        let wal = if wal_files.is_empty() {
76            // New database — create the first WAL file.
77            let wal_path = next_wal_path(&config.path, lsn_counter);
78            WalWriter::create(&wal_path, &config)?
79        } else {
80            let latest_path = wal_files.last().unwrap();
81            let opened = WalWriter::open(latest_path, &config)?;
82
83            let now_secs = std::time::SystemTime::now()
84                .duration_since(std::time::UNIX_EPOCH)
85                .unwrap_or_default()
86                .as_secs();
87
88            if opened.needs_rotation(now_secs) {
89                let new_lsn = lsn_counter + 1;
90                let new_path = next_wal_path(&config.path, new_lsn);
91                WalWriter::create(&new_path, &config)?
92            } else {
93                opened
94            }
95        };
96
97        let segment_store =
98            crate::segment::SegmentStore::open(config.path.clone(), config.cohort_window_secs)?;
99
100        Ok(Engine {
101            config,
102            wal,
103            memtable,
104            lsn_counter,
105            txid_counter,
106            lockfile,
107            segment_store,
108            snapshot_registry: crate::snapshot::SnapshotRegistry::new(),
109        })
110    }

119    pub fn put(&mut self, ns: &[u8], key: &[u8], val: &[u8]) -> Result<Lsn, EdgestoreError> {
120        if ns.len() > u16::MAX as usize {
121            return Err(EdgestoreError::NamespaceTooLong {
122                len: ns.len(),
123                max: u16::MAX as usize,
124            });
125        }
126
127        self.lsn_counter += 1;
128        let lsn = self.lsn_counter;
129        let timestamp = Self::now_nanos();
130
131        let record = WalRecord {
132            txid: 0,
133            lsn,
134            timestamp,
135            ttl: 0,
136            ns_len: ns.len() as u16,
137            ns_bytes: ns.to_vec(),
138            key_bytes: key.to_vec(),
139            op: Operation::Put,
140            value_hash: blake3::hash(val).into(),
141            value_bytes: val.to_vec(),
142        };
143        self.wal.append(&record)?;
144        self.rotate_wal_if_needed()?;
145
146        let encoded_key = encode_key(ns, key);
147        let entry = MemEntry {
148            key: encoded_key.clone(),
149            value: Some(val.to_vec()),
150            op: Operation::Put,
151            lsn,
152            timestamp,
153            ttl: 0,
154        };
155        self.memtable.insert(encoded_key, entry);
156
157        if (self.memtable.len() as u64) * AVG_ENTRY_SIZE_ESTIMATE >= self.config.segment_size_bytes {
158            let _ = self.flush_to_segments();
159        }
160
161        Ok(lsn)
162    }

211    /// Lazy expiry: records inserted with `put_with_ttl` are returned until `compact_once` removes their cohort.
212    pub fn get(&self, ns: &[u8], key: &[u8]) -> Result<Option<Vec<u8>>, EdgestoreError> {
213        let encoded_key = encode_key(ns, key);
214        match self.memtable.get(&encoded_key) {
215            Some(entry) if entry.op == Operation::Delete => return Ok(None),
216            Some(entry) => return Ok(entry.value.clone()),
217            None => {}
218        }
219        if let Some(entry) = self.segment_store.get(&encoded_key)? {
220            if entry.op == Operation::Delete {
221                return Ok(None);
222            }
223            return Ok(entry.value);
224        }
225        Ok(None)
226    }

262    /// Rotate the WAL if the current writer has exceeded `wal_max_bytes` or `wal_max_age_secs`.
263    ///
264    /// Called after every append so long-running sessions rotate inline without requiring
265    /// a close/reopen cycle.
266    fn rotate_wal_if_needed(&mut self) -> Result<(), EdgestoreError> {
267        let now_secs = SystemTime::now()
268            .duration_since(UNIX_EPOCH)
269            .unwrap_or_default()
270            .as_secs();
271        if !self.wal.needs_rotation(now_secs) {
272            return Ok(());
273        }
274        self.wal.fsync()?;
275        let new_path = next_wal_path(&self.config.path, self.lsn_counter);
276        self.wal = WalWriter::create(&new_path, &self.config)?;
277        Ok(())
278    }

280    /// Lazy expiry: TTL-expired records appear in range results until compaction removes their cohort.
281    pub fn range(
282        &self,
283        ns: &[u8],
284        start: &[u8],
285        end: &[u8],
286    ) -> Result<KvPairs, EdgestoreError> {
287        let enc_start = encode_key(ns, start);
288        let enc_end = encode_key(ns, end);
289
290        let mut merged: std::collections::HashMap<Vec<u8>, crate::types::MemEntry> =
291            std::collections::HashMap::new();
292
293        for (k, entry) in self.segment_store.range_scan(&enc_start, &enc_end)? {
294            merged.insert(k, entry);
295        }
296
297        for (k, entry) in self.memtable.range(&enc_start, &enc_end) {
298            let k_vec = k.to_vec();
299            let existing_lsn = merged.get(&k_vec).map(|e| e.lsn).unwrap_or(0);
300            if entry.lsn >= existing_lsn {
301                if entry.op == Operation::Delete {
302                    merged.remove(&k_vec);
303                } else {
304                    merged.insert(k_vec, entry.clone());
305                }
306            }
307        }
308
309        let mut keys: Vec<Vec<u8>> = merged.keys().cloned().collect();
310        keys.sort();
311        let mut out = Vec::new();
312        for k in keys {
313            if let Some(entry) = merged.get(&k) {
314                if entry.op == Operation::Delete { continue; }
315                if let Some(val) = &entry.value {
316                    let (_, raw_key) = decode_key(&k)?;
317                    out.push((raw_key, val.clone()));
318                }
319            }
320        }
321        Ok(out)
322    }

324    /// Lazy expiry: TTL-expired records appear in prefix results until compaction removes their cohort.
325    pub fn prefix(
326        &self,
327        ns: &[u8],
328        prefix: &[u8],
329    ) -> Result<KvPairs, EdgestoreError> {
330        let enc_prefix = encode_key(ns, prefix);
331
332        let mut merged: std::collections::HashMap<Vec<u8>, crate::types::MemEntry> =
333            std::collections::HashMap::new();
334
335        let seg_results = if let Some(enc_end) = prefix_upper_bound(&enc_prefix) {
336            self.segment_store.range_scan(&enc_prefix, &enc_end)?
337                .into_iter()
338                .filter(|(k, _)| k.starts_with(&enc_prefix))
339                .collect::<Vec<_>>()
340        } else {
341            vec![]
342        };
343
344        for (k, entry) in seg_results {
345            merged.insert(k, entry);
346        }
347
348        for (k, entry) in self.memtable.prefix(&enc_prefix) {
349            let k_vec = k.to_vec();
350            let existing_lsn = merged.get(&k_vec).map(|e| e.lsn).unwrap_or(0);
351            if entry.lsn >= existing_lsn {
352                if entry.op == Operation::Delete {
353                    merged.remove(&k_vec);
354                } else {
355                    merged.insert(k_vec, entry.clone());
356                }
357            }
358        }
359
360        let mut keys: Vec<Vec<u8>> = merged.keys().cloned().collect();
361        keys.sort();
362        let mut out = Vec::new();
363        for k in keys {
364            if let Some(entry) = merged.get(&k) {
365                if entry.op == Operation::Delete { continue; }
366                if let Some(val) = &entry.value {
367                    let (_, raw_key) = decode_key(&k)?;
368                    out.push((raw_key, val.clone()));
369                }
370            }
371        }
372        Ok(out)
373    }

375    pub fn flush_to_segments(&mut self) -> Result<crate::types::SegmentMeta, EdgestoreError> {
376        if self.memtable.is_empty() {
377            return Err(EdgestoreError::SegmentCorrupt("memtable is empty".to_string()));
378        }
379        let meta = self.segment_store.flush_memtable(self.memtable.as_ref())?;
380        self.memtable.clear();
381        Ok(meta)
382    }

393    pub fn commit_transaction(&mut self, tx: crate::transaction::Transaction) -> Result<Lsn, EdgestoreError> {
394        let mut tx = tx;
395        let records = tx.take_pending()?;
396        let mut last_lsn = self.lsn_counter;
397
398        for mut record in records {
399            self.lsn_counter += 1;
400            record.lsn = self.lsn_counter;
401            last_lsn = self.lsn_counter;
402
403            self.wal.append(&record)?;
404
405            let encoded_key = encode_key(&record.ns_bytes, &record.key_bytes);
406            let entry = MemEntry {
407                key: encoded_key.clone(),
408                value: if record.op == Operation::Put {
409                    Some(record.value_bytes.clone())
410                } else {
411                    None
412                },
413                op: record.op,
414                lsn: record.lsn,
415                timestamp: record.timestamp,
416                ttl: record.ttl,
417            };
418            self.memtable.insert(encoded_key, entry);
419        }
420
421        self.wal.fsync()?;
422        self.rotate_wal_if_needed()?;
423        Ok(last_lsn)
424    }

430    /// Run one bounded compaction cycle.
431    ///
432    /// Uses wall-clock time to determine which cohorts are expired.
433    /// Respects the write budget from `EdgestoreConfig::compaction_write_budget_bytes`.
434    /// Pinned segments (held by live snapshots) are never removed or rewritten.
435    ///
436    /// After compaction, the segment store is reloaded from disk so subsequent
437    /// reads see the updated segment list.
438    pub fn compact_once(&mut self) -> Result<crate::compactor::CompactionStats, EdgestoreError> {
439        let now_nanos = SystemTime::now()
440            .duration_since(UNIX_EPOCH)
441            .unwrap_or_default()
442            .as_nanos() as i64;
443        let pinned = self.snapshot_registry.pinned_ids();
444        let compactor = crate::compactor::Compactor::new(
445            self.config.path.clone(),
446            self.config.compaction_write_budget_bytes,
447            self.config.cohort_window_secs,
448        );
449        let mut manifest = crate::manifest::Manifest::open(&self.config.path.join("manifest.mf"))?;
450        let stats = compactor.compact_cycle(&mut manifest, now_nanos, &pinned)?;
451        // Reload segment_store so Engine sees the updated segment list after compaction.
452        self.segment_store =
453            crate::segment::SegmentStore::open(self.config.path.clone(), self.config.cohort_window_secs)?;
454        Ok(stats)
455    }

457    /// Return a point-in-time snapshot pinning the current set of segments.
458    ///
459    /// The returned `Snapshot` holds a reference to the `SnapshotRegistry` and
460    /// releases its pins automatically when dropped.
461    pub fn snapshot(&self) -> Result<crate::snapshot::Snapshot, EdgestoreError> {
462        let ids = self.segment_store.segment_ids();
463        let sid = self.snapshot_registry.register(&ids);
464        Ok(crate::snapshot::Snapshot::new(
465            sid,
466            self.snapshot_registry.clone(),
467            ids,
468            self.config.path.clone(),
469        ))
470    }
