32/// Applies a finalization mix to a randomly-seeded key, resulting in an avalanched hash. This
33/// helps avoid high false-positive ratios (see Section 4 in the paper).
34#[inline]
35pub const fn mix(key: u64, seed: u64) -> u64 {
36    murmur3::mix64(key.overflowing_add(seed).0)
37}

66/// Creates a block of sets, each set being of type T.

78/// Creates a block to store output fingerprints.
79/// This is distinguished from `make_block`, as we may want to randomize the unused fingerprints
80/// rather than making them all 0.
81///
82/// ## Why?
83///
84/// Inevitably some fingerprint entries will not be used. If all of these unused entries are 0,
85/// then the false-positive rate for a element x where fingerprint(x) = 0 is significantly higher
86/// than if the unused entries are uniformly random
87///
88/// Of course, the tradeoff here is that generating random elements is more expensive than
89/// memsetting a bunch of zeroes, so the option is configurable with the `uniform-random` feature.
90#[doc(hidden)]
91#[macro_export]
92macro_rules! make_fp_block(
93    ($size:ident) => {
94        {
95            #[cfg(feature = "uniform-random")] {
96                use rand::Rng;
97                let mut rng = rand::thread_rng();
98                let mut block = Vec::with_capacity($size);
99                for _ in 0..$size {
100                    block.push(rng.gen());
101                }
102                block.into_boxed_slice()
103            }
104
105            #[cfg(not(feature = "uniform-random"))] {
106                make_block!(with $size sets)
107            }
108        }
109    }
110);

112/// Creates a block of sets, each set being of type T.
113#[doc(hidden)]
114#[macro_export]
115macro_rules! try_enqueue(
116    (block $H_block:expr, set $idx:ident; queue block $Q_block:expr, with size $qblock_size:expr) => {
117        if $H_block[$idx].count == 1 {
118            $Q_block[$qblock_size].index = $idx;
119            // If there is only one key, the mask contains it wholly.
120            $Q_block[$qblock_size].hash = $H_block[$idx].mask;
121            $qblock_size += 1;
122        }
123    };
124);

126/// Checks if a collection of keys has all distinct values.
127#[cfg(debug_assertions)]
128pub fn all_distinct(keys: impl IntoIterator<Item = u64>) -> bool {
129    let mut s = alloc::collections::BTreeSet::new();
130    keys.into_iter().all(move |x| s.insert(x))
