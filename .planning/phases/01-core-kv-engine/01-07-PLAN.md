pub use edgestore::Engine;
pub use edgestore::EdgestoreConfig;
pub use edgestore::EdgestoreError;
pub use edgestore::Transaction;

impl Engine {
    pub fn open(config: EdgestoreConfig) -> Result<Engine, EdgestoreError>;
    pub fn put(&mut self, ns: &[u8], key: &[u8], val: &[u8]) -> Result<Lsn, EdgestoreError>;
    pub fn put_with_ttl(&mut self, ns: &[u8], key: &[u8], val: &[u8], ttl_secs: u32) -> Result<Lsn, EdgestoreError>;
    pub fn get(&self, ns: &[u8], key: &[u8]) -> Result<Option<Vec<u8>>, EdgestoreError>;
    pub fn delete(&mut self, ns: &[u8], key: &[u8]) -> Result<Lsn, EdgestoreError>;
    pub fn range(&self, ns: &[u8], start: &[u8], end: &[u8]) -> Result<Vec<(Vec<u8>, Vec<u8>)>, EdgestoreError>;
    pub fn prefix(&self, ns: &[u8], prefix: &[u8]) -> Result<Vec<(Vec<u8>, Vec<u8>)>, EdgestoreError>;
    pub fn flush(&mut self) -> Result<(), EdgestoreError>;
    pub fn begin(&mut self) -> Transaction;
    pub fn commit_transaction(&mut self, tx: Transaction) -> Result<Lsn, EdgestoreError>;
    pub fn rollback_transaction(&mut self, tx: Transaction);
}

    // Convenience wrappers (CORE-06): delegate to engine methods

    pub fn rollback(self, engine: &mut Engine);
}
