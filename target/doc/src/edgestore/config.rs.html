1use std::path::PathBuf;
2use crate::types::Compression;
3use crate::memtable::{MemTable, BTreeMemTable};

5/// Database configuration.
6pub struct EdgestoreConfig {
7    /// Filesystem path for the database directory.
8    pub path: PathBuf,
9    /// Maximum WAL file size in bytes before rotation.
10    pub wal_max_bytes: u64,
11    /// Maximum WAL file age in seconds before rotation.
12    pub wal_max_age_secs: u64,
13    /// Target uncompressed memtable size in bytes before flush.
14    pub segment_size_bytes: u64,
15    /// Cohort window width in seconds for deathtime compaction.
16    pub cohort_window_secs: u64,
17    /// Compression algorithm for WAL records.
18    pub compression_wal: Compression,
19    /// Compression algorithm for segment blocks.
20    pub compression_segments: Compression,
21    /// Target false-positive rate for xor filters.
22    pub xor_filter_fpr: f64,
23    /// Write-amplification budget per compaction cycle in bytes.
24    pub compaction_write_budget_bytes: u64,
25    /// Enable FDP (Flexible Data Placement) hints on NVMe 2.0 hardware.
26    pub fdp_enabled: bool,
27    /// Factory function that returns a new empty memtable.
28    pub memtable_factory: Box<dyn Fn() -> Box<dyn MemTable> + Send + Sync>,
29}

31impl std::fmt::Debug for EdgestoreConfig {
32    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
33        f.debug_struct("EdgestoreConfig")
34            .field("path", &self.path)
35            .field("wal_max_bytes", &self.wal_max_bytes)
36            .field("wal_max_age_secs", &self.wal_max_age_secs)
37            .field("segment_size_bytes", &self.segment_size_bytes)
38            .field("cohort_window_secs", &self.cohort_window_secs)
39            .field("compression_wal", &self.compression_wal)
40            .field("compression_segments", &self.compression_segments)
41            .field("xor_filter_fpr", &self.xor_filter_fpr)
42            .field("compaction_write_budget_bytes", &self.compaction_write_budget_bytes)
43            .field("fdp_enabled", &self.fdp_enabled)
44            .field("memtable_factory", &"<fn>")
45            .finish()
46    }
47}

49impl EdgestoreConfig {
50    /// Create a new `EdgestoreConfig` with sensible defaults.
51    pub fn new(path: impl Into<PathBuf>) -> Self {
52        EdgestoreConfig {
53            path: path.into(),
54            wal_max_bytes: 64 * 1024 * 1024,
55            wal_max_age_secs: 60,
56            segment_size_bytes: 16 * 1024 * 1024,
57            cohort_window_secs: 3600,
58            compression_wal: Compression::Lz4,
59            compression_segments: Compression::Zstd(1),
60            xor_filter_fpr: 0.01,
61            compaction_write_budget_bytes: 256 * 1024 * 1024,
62            fdp_enabled: false,
63            memtable_factory: Box::new(|| Box::new(BTreeMemTable::new())),
64        }
65    }
66}
