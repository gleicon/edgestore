10///
11/// See <http://creativecommons.org/publicdomain/zero/1.0/>.
12pub const fn splitmix64(seed: &mut u64) -> u64 {
13    *seed = (*seed).overflowing_add(0x9e37_79b9_7f4a_7c15).0;
14    let mut z = *seed;
15    z = (z ^ (z >> 30)).overflowing_mul(0xbf58_476d_1ce4_e5b9).0;
16    z = (z ^ (z >> 27)).overflowing_mul(0x94d0_49bb_1331_11eb).0;
17    z ^ (z >> 31)
18}
